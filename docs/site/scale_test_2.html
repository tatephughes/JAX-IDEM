import pandas as pd
import plotly.express as px

# Read the CSV into a DataFrame
df = pd.read_csv("data/scale_3.csv")

# Keep only entries with the specified conditions
df_filtered = df.query("n == 3 and d == 2 and is_64_bit == True").sort_values("T")

# Create a line plot for each filter type
fig = px.line(
    df_filtered,
    x="T",
    y="average_time",     # Or "compile_time" if preferred
    color="Filter",         # Differentiates filter lines
    markers=True,         # Optional: adds dots to each T value
    title="Average Compute Time vs T for Filters (n=3, d=2, 64-bit)",
    labels={
        "T": "Time Horizon (T)",
        "average_time": "Avg. Compute Time (s)",
        "name": "Filter Type"
    }
)

fig.show()
