1 Standard RMH

Here is a very simple example of a RMH algorithm targetting a simple 2D Gaussian

Code 
import jax
jax.config.update("jax_enable_x64", True)
import jax.numpy as jnp
import jax.random as rand

from jax.numpy.linalg import inv, outer

import time

from jax.scipy.stats.multivariate_normal import logpdf as logmvn
from jax.random import multivariate_normal as smvn

d = 100

target_mean = jnp.ones(d)

#target_cov = jnp.array([[1,0.5],[0.5,1]])
key = jax.random.PRNGKey(seed=60)
M = rand.normal(key, shape = (d,d))
target_cov = M.T@M


def log_target(state):
    #useless_matrix = outer(state, state) + inv(10*jnp.identity(d))
    #id = inv(useless_matrix) @ useless_matrix
    return logmvn(state, target_mean, target_cov)

init_state = jnp.ones(d)
#prop_cov = jnp.eye(d)/jnp.sqrt(d)

# Theoretically 'optimal'
prop_cov = target_cov * (5.6644/d)


import jaxidem.utils as utils
#time_res = utils.time_jit(key, log_target, init_state, 1000)

n = 1000

@jax.jit
def rmh_step(state, key):
    prop_key, acc_key = jax.random.split(key, 2)

    proposal = smvn(prop_key, state, prop_cov)
    r = log_target(proposal) - log_target(state)

    log_prob = jnp.minimum(0.0, r)

    u = jax.random.uniform(acc_key)

    new_state = jax.lax.cond((jnp.log(u) < log_prob),
                             0, lambda _: proposal,
                             0, lambda _: state)

    return new_state, new_state



key = jax.random.PRNGKey(42)
# ensure compilation
print(rmh_step(init_state, key)[0][0], flush=True)
keys = jax.random.split(key, 2*n)

start = time.time()
_, chain_1 = jax.lax.scan(rmh_step, init_state, keys[:n])
X_true = jnp.vstack([init_state, chain_1])
end = time.time()

seq_time = end-start

print(f"\nStandard sequention RMH took {seq_time:.6f} seconds")

res_mean = jnp.mean(chain_1, axis=0)
res_cov = jnp.cov(chain_1.T)

#print(res_mean)

accepted = jnp.any(X_true[1:] != X_true[:-1], axis=1)
acceptance_rate = jnp.mean(accepted)

2 Picard Map

General Markov chains on the domain \(\mathcal{X}\) can be written as

\[\begin{split} \bv X_{i+1} = g(\bv X_i, W_i),\quad i=0,1,\dots \end{split} \tag{1}\]

Where \(X^{(0)}\in\mathcal{X}\) is fixed, \(X^{(i)}\in\mathcal{X}\) and \(W_i\in\mathcal{W}\) are random states (in JAX, the keys serve this purpose).

We can rewrite Equation 1 as a random-step type markov chain form like so;

\[\begin{split} \bv X_{i+1} = \bv X_{i} + \underbrace{\left(g(\bv X_{i}, W_i) - \bv X_{i}\right)}_{f(\bv X_{i}, W_i)}. \end{split} \tag{2}\]

This is now in the form used by Grazzi and Zanella (2025), so we can define the Picard map \(\Phi: \mathcal{X}^{K+1}\times\mathcal{W}^K \mapsto\mathcal{X}^{K+1}\);

\[\begin{split} \bv X_{i}' = \Phi_i(X_i, W) = \begin{cases} \bv X_0 &\quad i=0\\ \bv X_0 + \sum_{l=0}^{i-1} f(X_l, W_l) &\quad 0<i<K. \end{cases} \end{split} \tag{3}\]

Now we define a function that takes a particular step function \(g(\bv X_i,W_i)\) and gives the picard map corresponding to \(f(\bv X_i,W_i) = g(\bv X_i, W_i) - \bv X_i\).

Code 
def get_picard_map(step):
    # vectorise the step function
    vf = jax.vmap(lambda s, key: step(s, key)[0] - s)

    @jax.jit
    def picard_map(states, keys):
        init_state = states[0]        
        fs = vf(states[0:-1], keys)
        cumsum = init_state + jnp.cumsum(fs, axis=0)
        return jnp.vstack([init_state, cumsum])
    return picard_map

Now we define a new function to apply the map only to a section of the chain, and return the index to which the chain has already converged.

3 JIT-Luc Picard

Code 
def jit_luc(pic_map, K, keys): 

    @jax.jit
    def online_step(tup):

        states, L, i = tup
        
        d = states.shape[1]

        states_to_update = jax.lax.dynamic_slice(states, (L,0), (K, d))
        keys_to_update_with = jax.lax.dynamic_slice(keys, (L,0), (K-1, 2))
        
        updated_states = pic_map(states_to_update, keys_to_update_with)

        states_new = jax.lax.dynamic_update_slice(states, updated_states, (L,0))

        states_incorrect = jnp.all(states_to_update != updated_states, axis=1)

        L_new = L + jnp.where(states_incorrect, size=1, fill_value=K-1)[0][0]
        
        return states_new, L_new, i+1
    
    return online_step


pic_map = get_picard_map(rmh_step)

K = 10
X_0 = jnp.stack([init_state for _ in range(n+K+1)])


jit_step = jit_luc(pic_map, K, keys)

L = 0
init_val = (X_0, jnp.array(L), 0)

start = time.time()
X_final, L, iterations = jax.lax.while_loop(lambda val: val[1]<n, jit_step, init_val)
end = time.time()

jitpar_time = end-start

4 Optimal K

I’m actually now interested in the optimal value for K. Assuming each markov step can only be done by one core at a time, the optimal K is probably simply the number of cores that the machine has, since that will exactly saturate the cpu. But here, with JAX, we mostly trust JAX to parallelise things, as a result, each RMH step is already trying to saturate the CPU (check htop).

This means that only so much more parallelisation can be added before we lose benefits, even for K below the number of cores. This matters because we don’t want to waste computation; if, during an online step, the algorithm finds that only, for example, the first 3 indices match, but K=10 so that 10 states are updates, then the last 7 computations where wasted, so K=3 would have been better.

So there is, presumably, an optimal K. I would postulate that the optimal K depends directly on the number of CPU cores as well as the acceptance rate; a lower acceptance rate means that more steps can be ‘skipped’, and so more indices match at each step, so ideally K would be exactly this number of indices. Since we probably dont want to eveluate the picard map up to an arbitrary point (at with the new states mismatch the last step), since that would make parallelism less obvious, K likely relates to either the average number of times that the chain rejects in a row, or perhaps the maximum number of times, for finite chains.

For the chain above, lets get the computation time and number of iterations for K=2,…,10.

(note that, in my implementation, theres a bit of a quirk; K-1 steps are actually computed at once, as opposed to K.)

5 Quick alternative

Code 
@jax.jit
def f(state, key):
    prop_key, acc_key = jax.random.split(key, 2)

    proposal = smvn(prop_key, state, prop_cov)
    r = log_target(proposal) - log_target(state)

    log_prob = jnp.minimum(0.0, r)

    u = jax.random.uniform(acc_key)

    accepted = jnp.log(u) < log_prob

    new_state = jax.lax.cond(accepted,
                             0, lambda _: proposal,
                             0, lambda _: state)

    return new_state - state, accepted

def jit_luc_tng(f, K, keys):
    
    vf = jax.vmap(lambda s, key: f(s, key))

    @jax.jit
    def online_step(tup):

        states, L, i = tup
        
        d = states.shape[1]

        states_to_update = jax.lax.dynamic_slice(states, (L,0), (K, d))
        keys_to_update_with = jax.lax.dynamic_slice(keys, (L,0), (K-1, 2))

        init_state = states_to_update[0]        
        fs, accepted = vf(states_to_update[0:-1], keys_to_update_with)
        updated_states = init_state + jnp.cumsum(fs, axis=0)

        states_new = jax.lax.dynamic_update_slice(states, updated_states, (L+1,0))

        states_incorrect = jnp.all(states_to_update[1:] != updated_states, axis=1)

        L_new = L + jnp.where(states_incorrect, size=1, fill_value=K-1)[0][0] + 1
        
        return states_new, L_new, i+1
    
    return online_step
