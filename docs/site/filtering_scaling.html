# creates frames to gridded data
def plotly_st_grid(st,
                   title="Spatio-Temporal Field",
                   colorscale="Viridis",
                   sort_axes=True,      # sort unique x, y, t; set False to keep first-seen order
                   show_colorbar=True,
                   frame_duration=100,  # ms
                   ):

    # adapted from (and probably stolen by) an llm. Probably rethink or find a proper attributation.
    
    x = st.x
    y = st.y
    z = st.z
    t = st.t

    def unique_axis(vals):
        if sort_axes:
            uniq = jnp.unique(vals)
            to_idx = lambda v: jnp.searchsorted(uniq, v)
        else:
            _, idx = jnp.unique(vals, return_index=True)
            order = jnp.sort(idx)
            uniq = vals[order]
            mapping = {v: i for i, v in enumerate(uniq.tolist())}
            to_idx = lambda v: jnp.fromiter((mapping[vi] for vi in v.tolist()), dtype=int)
        return uniq, to_idx

    ux, xi = unique_axis(x)
    uy, yi = unique_axis(y)
    ut, ti = unique_axis(t)

    if callable(xi):
        xi = xi(x)
        yi = yi(y)
        ti = ti(t)

    nx, ny, nt = len(ux), len(uy), len(ut)

    z_grid = jnp.full((nt, ny, nx), jnp.nan, dtype=float)
    z_grid = z_grid.at[ti, yi, xi].set(z)

    zmin = float(jnp.nanmin(z_grid))
    zmax = float(jnp.nanmax(z_grid))
    
    frames = []
    for i in range(nt):
        frames.append(
            go.Frame(
                name=f"t = {ut[i]}",
                data=[
                    go.Heatmap(
                        z=z_grid[i],
                        x=ux,
                        y=uy,
                        colorscale=colorscale,
                        zmin=zmin,
                        zmax=zmax,
                        colorbar=dict(title="z") if show_colorbar else None,
                        showscale=show_colorbar,
                    )
                ],
            )
        )

    return frames

2 Filters with Likelihood computation

2.1 Standard Kalman Filter

from jax.scipy.linalg import solve_triangular as st

@jax.jit
def kalman_filter(m_0, P_0, M, PHI, S2_eta, S2_eps, zs):

    r = m_0.shape[0]
    nobs = zs.shape[1]
    
    @jax.jit
    def step(carry, z_t):
        m_tt, P_tt, _, _, ll = carry

        # predict
        m_pred = M @ m_tt
        P_pred = M @ P_tt @ M.T + S2_eta
        # Update
        e_t = z_t - PHI @ m_pred
        Sigma_t = PHI @ P_pred @ PHI.T + S2_eps
        K_t = (jsc.linalg.solve(Sigma_t, PHI, assume_a = "pos") @ P_pred.T).T 
        
        m_up = m_pred + K_t @ e_t
        P_up = (jnp.eye(r) - K_t @ PHI) @ P_pred
        
        # Stabilised form
        # P_up = (jnp.eye(r) - K_t @ PHI) @ P_pred @ (jnp.eye(r) - K_t @ PHI).T + K_t@S2_eps@K_t.T

        # likelihood of epsilon, using cholesky decomposition
        chol_Sigma_t = jnp.linalg.cholesky(Sigma_t)
        z = st(chol_Sigma_t, e_t, lower=True)
        ll_new = (
            ll
            - jnp.sum(jnp.log(jnp.diag(chol_Sigma_t)))
            - 0.5 * jnp.dot(z, z)
            - 0.5 * nobs * jnp.log(2 * jnp.pi)
        )

        new_carry = (m_up, P_up, m_pred, P_pred, ll_new)
        return new_carry, new_carry


    carry, seq = jl.scan(
        step,
        (m_0, P_0, m_0, P_0, 0),
        zs,
    )

    ll, ms, Ps, mpreds, Ppreds = (carry[4], seq[0], seq[1], seq[2], seq[3])
    
    filt_results = {"ll": ll,
                    "ms": ms,
                    "Ps": Ps,
                    "mpreds": mpreds,
                    "Ppreds": Ppreds}

    return filt_results
m_0 = jnp.zeros(4)
P_0 = 100*jnp.eye(4)

filt_results_kal = kalman_filter(m_0, P_0, M, PHI, S2_eta, S2_eps, zs)

print(f"Log-Likelihood = {filt_results_kal['ll']}")

2.2 Information Filter

@jax.jit
def information_filter(
        nu_0,
        Q_0,
        M,
        PHI,
        S2_eta,
        S2_eps,
        zs
):

    r = nu_0.shape[0]
    nobs = zs.shape[1]

    i = jax.vmap(lambda z: PHI.T @ jsc.linalg.solve(S2_eps, z, assume_a = "pos"), in_axes=0)(zs)
    I = PHI.T @ jsc.linalg.solve(S2_eps, PHI, assume_a = "pos")
    
    Minv = jnp.linalg.inv(M)
    S2_eta_inv = jnp.linalg.inv(S2_eta)

    @jax.jit
    def step(carry, i):
        nu_tt, Q_tt, _, _ = carry

        S_t = Minv.T @ Q_tt @ Minv
        J_t = jsc.linalg.solve((S_t + S2_eta_inv).T, S_t.T).T
        
        nu_pred = (jnp.eye(r) - J_t) @ Minv.T @ nu_tt
        Q_pred = (jnp.eye(r) - J_t) @ S_t

        nu_up = nu_pred + i
        Q_up = Q_pred + I

        new_carry = (nu_up, Q_up, nu_pred, Q_pred)
        
        return new_carry, new_carry
    
    carry, seq = jl.scan(
        step,
        (nu_0, Q_0, jnp.zeros(r), jnp.eye(r)),
        i,
    )
    
    nus, Qs, nupreds, Qpreds = (seq[0], seq[1], seq[2], seq[3])

    @jax.jit
    def likelihood_func(z, nu_pred, Q_pred):
        e = z - PHI @ jnp.linalg.solve(Q_pred, nu_pred)
        chol_Sigma_t = jnp.linalg.cholesky(
            PHI @ jnp.linalg.solve(Q_pred, PHI.T) + S2_eps
        )
        z = st(chol_Sigma_t, e, lower=True)

        ll = (
            -jnp.sum(jnp.log(jnp.diag(chol_Sigma_t)))
            - 0.5 * jnp.dot(z, z)
            - 0.5 * nobs * jnp.log(2 * jnp.pi)
        )
        return ll

    lls = jax.vmap(likelihood_func)(zs, nupreds, Qpreds)
    ll = jnp.sum(lls)
    
    filt_results = {"ll": ll,
                    "nus": nus,
                    "Qs": Qs,
                    "nupreds": nupreds,
                    "Qpreds": Qpreds}

    return filt_results

2.3 Square-root filter

@jax.jit
def qr_R(A, B):
    """Wrapper for the stacked-QR decompositon"""
    return jnp.linalg.qr(jnp.vstack([A, B]), mode="r")

@jax.jit
def sqrt_filter(m_0, U_0, M, PHI, S2_eta, S2_eps, zs):

    r = m_0.shape[0]
    nobs = zs.shape[1]

    S_eta = jnp.linalg.cholesky(S2_eta, upper=True)
    S_eps = jnp.linalg.cholesky(S2_eps, upper=True)

    @jax.jit
    def step(carry, z_t):
        m_tt, U_tt, _, _, ll= carry

        # predict
        m_pred = M @ m_tt
        U_pred = qr_R(U_tt @ M.T, S_eta)
        #U_pred = jnp.linalg.cholesky(M @ U_tt.T @U_tt @ M.T + S2_eta, upper=True)
        # Update
        e_t = z_t - PHI @ m_pred
        Ui_t = qr_R(U_pred @ PHI.T, S_eps)
        #Ui_t = jnp.linalg.cholesky(PHI @ U_pred.T@U_pred @ PHI.T + S2_eps, upper=True)

        # Kalman Gain2
        K_t =(
            st(
                Ui_t,
                st(Ui_t.T, PHI, lower=True)
                @ U_pred.T
                @ U_pred,
            )
        ).T

        m_up = m_pred + K_t @ e_t

        U_up = qr_R(U_pred @ (jnp.eye(r) - K_t @ PHI).T, S_eps @ K_t.T)
        s = st(Ui_t.T, e_t, lower=True)
 
        ll_new = (ll
                  - jnp.sum(jnp.log(jnp.abs(jnp.diag(Ui_t))))
                  - 0.5 * jnp.dot(s, s)
                  - 0.5 * nobs * jnp.log(2 * jnp.pi)
                  )

        new_carry = (m_up, U_up, m_pred, U_pred, ll_new)
        return new_carry, new_carry

    carry, seq = jl.scan(
        step,
        (m_0, U_0, m_0, U_0, 0),
        zs,
    )

    ll, ms, Us, mpreds, Upreds = (carry[4], seq[0], seq[1], seq[2], seq[3])

    filt_results = {"ll": ll,
                    "ms": ms,
                    "Us": Us,
                    "mpreds": mpreds,
                    "Upreds": Upreds,}

    return filt_results
U_0 = jnp.linalg.cholesky(P_0)

2.4 Sqrt-information filter

@jax.jit
def ql_L(A, B):
    """Wrapper for the stacked-QL decompositon"""
    A_flipped = jnp.flip(jnp.vstack([A, B]), axis=1)
    R = jnp.linalg.qr(A_flipped, mode='r')
    L = jnp.flip(R, axis=(0, 1))

    return L


def safe_cholesky(matrix):
    def zero_case(_):
        return jnp.zeros_like(matrix)
    def nonzero_case(matrix):
        return jnp.linalg.cholesky(matrix, upper=True)
    return jax.lax.cond(
        jnp.all(matrix == 0),
        zero_case,
        nonzero_case,
        operand=matrix
        )


@jax.jit
def sqinf_filter(nu_0, R_0, M, PHI, S2_eta, S2_eps, zs):

    nobs = zs.shape[1]

    S_eps, _ = jsc.linalg.cho_factor(S2_eps, lower = False)
    S_eta, _ = jsc.linalg.cho_factor(S2_eta, lower = False)

    # pretty quick
    i_ts = jax.vmap(lambda z: PHI.T @ jsc.linalg.cho_solve((S_eps, True), z), in_axes=0)(zs)
    R = jnp.linalg.qr(st(S_eps.T, PHI, lower=False), mode="r")
    #R = safe_cholesky(PHI.T @ jsc.linalg.cho_solve((S_eps, True), PHI))

    @jax.jit
    def step(carry, i):
        nu_tt, R_tt, _, _ = carry

        U_pred = ql_L(st(R_tt.T, M.T, lower=True), S_eta)
        R_pred = st(U_pred, jnp.eye(r), lower=True).T
        nu_pred = R_pred.T @ R_pred @ M @ st(R_tt,
                                             st(R_tt.T,
                                                nu_tt,
                                                lower=True), lower=False)

        nu_up = nu_pred + i
        R_up = qr_R(R_pred, R)
        #R_up = jnp.linalg.cholesky(R_pred.T @R_pred + R.T@R, upper=True)

        new_carry = (nu_up, R_up, nu_pred, R_pred)
        return new_carry, new_carry

    # also pretty quick
    carry, seq = jl.scan(
        step,
        (nu_0, R_0, jnp.zeros(r), jnp.eye(r)),
        i_ts,
    )

    nus, Rs, nupreds, Rpreds = (seq[0], seq[1], seq[2], seq[3])

    # THIS is where slowdown happens. why?
    def likelihood_func(z, nu_pred, R_pred):

        # IS IT THE qr_Rs fault????

        Ui_t = jnp.linalg.cholesky(PHI @ st(R_pred, st(R_pred.T, PHI.T, lower=True)) + S2_eps)
        s = st(Ui_t, e, lower=True)

        ll = (
            - jnp.sum(jnp.log(jnp.abs(jnp.diag(Ui_t))))
            - 0.5 * jnp.dot(s, s)
            - 0.5 * nobs * jnp.log(2 * jnp.pi)
        )
        
        return ll
        
    lls = jax.vmap(likelihood_func)(zs, nupreds, Rpreds)
    ll = jnp.sum(lls)
    

    filt_results = {"ll": ll,
                    "nus": nus,
                    "Rs": Rs,

@jax.jit
def pkalman_filter(m_0, P_0, M, PHI, S2_eta, S2_eps, zs):
    
    r = m_0.shape[0]
    nobs = zs.shape[1]

    m1pred = M @ m_0
    P1pred = M @ P_0 @ M.T + S2_eta

    S_1 =  PHI @ P1pred @ PHI.T + S2_eps

    # cholS = jsc.linalg.cho_factor(S_1)
    #D_1 = (jsc.linalg.cho_solve(cholS, PHI_tree[0])@P1pred.T).T

    D_1 = (jnp.linalg.solve(S_1, PHI @ P1pred)).T
    
    A_1  = jnp.zeros((r,r))
    b_1  = m1pred + D_1 @ (zs[0] - PHI @ m1pred)
    C_1  = P1pred - D_1 @ S_1 @ D_1.T

    # this is how it is in the paper. Adrien Corenflos has a slighlty different
    # equation, using the expression for S_k here. Not sure why the discrepency is here.
    
    nu_1 = M.T @ PHI.T @ jnp.linalg.solve(S_1, zs[0])
    Q_1  = M.T @ PHI.T @ jnp.linalg.solve(S_1, PHI) @ M

    first_elt = (A_1, b_1, C_1, nu_1, Q_1)

    @jax.jit
    def get_element(z_k):
                
        S_k = PHI @ S2_eta @ PHI.T + S2_eps
                
        D_k = (jsc.linalg.solve(S_k, PHI @ S2_eta)).T

        imdp = (jnp.eye(r) - D_k @ PHI)
        
        A_k = imdp @ M
        b_k = D_k @ z_k
        
        C_k = imdp @ S2_eta

        nu_k = M.T @ PHI.T @ jsc.linalg.solve(S_k, z_k)
        Q_k = M.T @ PHI.T @ jsc.linalg.solve(S_k, PHI @ M)
    
        return (A_k, b_k, C_k, nu_k, Q_k)

    elts = jax.vmap(get_element)(zs[1:])

    all_elts = jax.tree.map(lambda f, e: jnp.concatenate([f[None, :], e]), first_elt, elts)

    @jax.vmap
    def compose(elt_i, elt_j):
        
        A_i, b_i, C_i, nu_i, Q_i = elt_i
        A_j, b_j, C_j, nu_j, Q_j = elt_j

        I = jnp.eye(r)
        
        ipcq = I + C_i @ Q_j
        ipqc = I + Q_j @ C_i

        # lots of cho solves, can be simplifies. 
        A_ij = A_j @ jsc.linalg.solve(ipcq, A_i)
        b_ij = A_j @ jsc.linalg.solve(ipcq, (b_i + C_i @ nu_j)) + b_j
        C_ij = A_j @ jsc.linalg.solve(ipcq, C_i @ A_j.T) + C_j
        nu_ij = A_i.T @ jsc.linalg.solve(ipqc, nu_j - Q_j@b_i) + nu_i
        Q_ij = A_i.T @ jsc.linalg.solve(ipqc, Q_j @ A_i) + Q_i
        
        return (A_ij, b_ij, C_ij, nu_ij, Q_ij)
    
    final_elts = jl.associative_scan(compose, all_elts)

    ms = final_elts[1]
    Ps = final_elts[2]

    nu, Q = final_elts[3][-1], final_elts[4][-1]

    @jax.jit
    def get_ll(z, m, P):

        m_pred = M @ m
        P_pred = M @ P @ M.T + S2_eta

        e = z - PHI @ m_pred
        S = PHI @ P_pred @ PHI.T + S2_eps
        
        cholS = jnp.linalg.cholesky(S)
        z = st(cholS, e, lower=True)
        ll = (
            - jnp.sum(jnp.log(jnp.diag(cholS)))
            - 0.5 * jnp.dot(z, z)
            - 0.5 * nobs * jnp.log(2 * jnp.pi)
        )

        return ll
    
    ll = jnp.sum(jax.vmap(get_ll)(zs,
                                  jnp.concatenate([m_0[None, :], ms])[:-1],
                                  jnp.concatenate([P_0[None, :, :], Ps])[:-1]))


    
    filt_results = {"ll": ll,
                    "ms": ms,
                    "Ps": Ps}

    return filt_results
